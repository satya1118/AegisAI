CREATE OR REPLACE VIEW executive_risk_summary AS
SELECT 
    COUNT(*) AS total_vendors,
    SUM(contract_value) AS total_revenue,
    SUM(CASE WHEN risk_tier = 'High' THEN 1 ELSE 0 END) AS high_risk_vendors,
    SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END) AS revenue_at_risk,
    ROUND(
        100.0 * SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END)
        / SUM(contract_value),
        2
    ) AS pct_revenue_at_risk
FROM vendors;

SELECT * FROM executive_risk_summary;
