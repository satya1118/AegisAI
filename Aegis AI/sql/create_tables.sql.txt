
TRUNCATE TABLE vendors;
COPY vendors
FROM 'C:/Users/gnaid/Desktop/vendors.csv'
DELIMITER ','
CSV HEADER;
SELECT COUNT(*) FROM vendors;

SELECT COUNT(*) AS total_vendors
FROM vendors;
 SELECT risk_tier,
       COUNT(*) AS vendor_count,
       ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS percentage
FROM vendors
GROUP BY risk_tier
ORDER BY vendor_count DESC;
SELECT 
    SUM(contract_value) AS total_revenue,
    SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END) AS revenue_at_risk,
    ROUND(
        100.0 * SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END)
        / SUM(contract_value),
        2
    ) AS pct_revenue_at_risk
FROM vendors;
SELECT category,
       ROUND(AVG(sla_breaches_90d), 2) AS avg_sla_breaches
FROM vendors
GROUP BY category
ORDER BY avg_sla_breaches DESC;
SELECT vendor_id,
       vendor_name,
       contract_value,
       sla_breaches_90d,
       compliance_flag
FROM vendors
WHERE risk_tier = 'High'
AND contract_value > 1000000
ORDER BY contract_value DESC;
SELECT risk_tier,
       COUNT(*) AS vendor_count,
       ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS percentage
FROM vendors
GROUP BY risk_tier
ORDER BY vendor_count DESC;
SELECT 
    SUM(contract_value) AS total_revenue,
    SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END) AS revenue_at_risk,
    ROUND(
        100.0 * SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END)
        / SUM(contract_value),
        2
    ) AS pct_revenue_at_risk
FROM vendors;
CREATE OR REPLACE VIEW executive_risk_summary AS
SELECT 
    COUNT(*) AS total_vendors,
    SUM(contract_value) AS total_revenue,
    SUM(CASE WHEN risk_tier = 'High' THEN 1 ELSE 0 END) AS high_risk_vendors,
    SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END) AS revenue_at_risk,
    ROUND(
        100.0 * SUM(CASE WHEN risk_tier = 'High' THEN contract_value ELSE 0 END)
        / SUM(contract_value),
        2
    ) AS pct_revenue_at_risk
FROM vendors;
SELECT * FROM executive_risk_summary;
